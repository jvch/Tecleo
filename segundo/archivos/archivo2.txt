//programa que muestra y oculta imágenes de acuerdo a la opción elegida
onEvent("dropdown", "change", function() {
    var escena = getProperty("dropdown", "value");
    if (escena == "Playa") {
        showElement("playaImg");
        hideElement("nieveImg");
    } else {
        showElement("nieveImg");
        hideElement("playaImg");
    }
});
