#Script que elige dos sustantivos y dos adjetivos para formar una oración
import random
noun1 = ['The dog', 'The cat', 'The monkey', 'The hippo']
noun2 = ['the room', 'the air', 'the light', 'the night']
adjetive1 = ['happy', 'sad', 'timid', 'howling']
adjective2 = ['dark', 'haunted', 'ghstly', 'cold']
print(random.choice(noun1), 'is', random.choice(adjective1), random.choice(noun2), 'is', random.choice(adjective2))
