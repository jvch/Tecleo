#Crea un punto en una posición aleatoria
#con un color aleatorio
import random
import turtle
t = turtle.Turtle()
x = random.randint(-120, 120)
y = random.randint(-120, 120)
r = random.randint(0, 255) #red
g = random.randint(0, 255) #green
b = random.randint(0, 255) #blue
t.goto(x, y)
t.color(r, g, b)
t.dot(25)
