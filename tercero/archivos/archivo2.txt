import turtle
import random
t = turtle.Turtle()
#creación de una función para dibujar un árbol
def tree(x, y):
    t.penup()
    t.goto(x, y)
    t.color('brown')
    t.width(20)
    t.setheading(90)
    t.pendown()
    t.forward(50)
    t.color('green')
    t.dot(50)
#ciclo que crea 20 árboles en posiciones aleatorias
for n in range(20):
    tree(random.randint(-190, 190), random.randint(-190, 190))
