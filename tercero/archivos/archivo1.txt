import random
import turtle
t = turtle.Turtle()
#función que dibuja un oso
#con 'size' se especifica el tamaño de la cabeza
#y de las orejas
def head(size):
    t.color('brown')
    t.dot(size)
    t.setheading(45)
    t.forward(size/2)
    t.dot(size/2)
    t.back(size/2)
    t.setheading(135)
    t.forward(size/2)
    t.dot(size/2)
#llamada a la función 'head'
head(100)
